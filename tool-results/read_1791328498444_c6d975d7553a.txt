    98→function unwrapUrl(u: string): string {
    99→  try {
   100→    const parsed = new URL(u, window.location.origin);
   101→    if (parsed.pathname === "/api/browse") {
   102→      const inner = parsed.searchParams.get("u");
   103→      if (inner) return inner;
   104→    }
   105→    if (parsed.protocol.startsWith("http")) return parsed.toString();
   106→    return u;
   107→  } catch {
   108→    return u;
   109→  }
   110→}
   111→
   112→const hostOf = (u: string) => {
   113→  try {
   114→    return new URL(u).hostname.replace(/^www\./, "");
   115→  } catch {
   116→    return u.slice(0, 40);
   117→  }
   118→};

   125→function findLabelText(el: HTMLElement, doc: Document): string {
   126→  try {
   127→    const id = el.getAttribute("id");
   128→    if (id) {
   129→      const l = doc.querySelector(`label[for="${CSS.escape(id)}"]`);
   130→      if (l?.textContent) return l.textContent.trim().slice(0, 90);
   131→    }
   132→    const aria = el.getAttribute("aria-label");
   133→    if (aria) return aria.slice(0, 90);
   134→    const ph = (el as HTMLInputElement).placeholder;
   135→    if (ph) return ph.slice(0, 90);
   136→    const wrap = el.closest("label");
   137→    if (wrap?.textContent) return wrap.textContent.trim().slice(0, 90);
   138→    const name = el.getAttribute("name");
   139→    if (name) return name.slice(0, 90);
   140→    // previous text node sibling
   141→    let prev = el.previousElementSibling;
   142→    for (let i = 0; i < 3 && prev; i++) {
   143→      const t = prev.textContent?.trim();
   144→      if (t) return t.slice(0, 90);
   145→      prev = prev.previousElementSibling as Element | null;
   146→    }
   147→  } catch {
   148→    /* noop */
   149→  }
   150→  return el.getAttribute("name") || (el as HTMLInputElement).type || "field";
   151→}

   153→function setNativeValue(el: HTMLInputElement | HTMLTextAreaElement | HTMLSelectElement, value: string) {
   154→  const proto =
   155→    el instanceof HTMLTextAreaElement
   156→      ? HTMLTextAreaElement.prototype
   157→      : el instanceof HTMLSelectElement
   158→        ? HTMLSelectElement.prototype
   159→        : HTMLInputElement.prototype;
   160→  const desc = Object.getOwnPropertyDescriptor(proto, "value");
   161→  try {
   162→    desc?.set?.call(el, value);
   163→  } catch {
   164→    el.value = value;
   165→  }
   166→  el.dispatchEvent(new Event("input", { bubbles: true }));
   167→  el.dispatchEvent(new Event("change", { bubbles: true }));
   168→}

   204→function matchBank(field: PageField, bank: BankEntry[]): BankEntry | null {
   205→  const hay = `${field.label} ${field.name} ${field.id}`;
   206→  for (const b of bank) {
   207→    if (b.re.test(hay)) {
   208→      if (field.tag === "textarea" && !b.textarea && b.key !== "address") continue;
   209→      return b;
   210→    }
   211→  }
   212→  return null;
   213→}

   217→export function CopilotBrowser({ open, onOpenChange, initialUrl, appName, appSlug }: CopilotBrowserProps) {
   218→  const { profile, docs, apps, downloadDoc, pushBrowserHistory, browserHistory, browserBookmarks, addBrowserBookmark, removeBrowserBookmark, clearBrowserHistory } = useCopilot();
   219→  const { toast } = useToast();
   220→
   221→  const [tabs, setTabs] = useState<BrowserTab[]>([]);
   222→  const [activeId, setActiveId] = useState<string>("");
   223→  const [reloadKeys, setReloadKeys] = useState<Record<string, number>>({});
   224→  const [panelOpen, setPanelOpen] = useState(false);
   225→  const [panelTab, setPanelTab] = useState<"screen" | "docs" | "answers">("screen");
   226→  const [reads, setReads] = useState<Record<string, ScreenRead>>({});
   227→  const [guide, setGuide] = useState<Record<string, string>>({});
   228→  const [guideBusy, setGuideBusy] = useState(false);
   229→  const [mapBusy, setMapBusy] = useState(false);
   230→  const [historyOpen, setHistoryOpen] = useState(false);
   231→  const [addr, setAddr] = useState("");
   232→
   233→  const iframeRefs = useRef<Record<string, HTMLIFrameElement | null>>({});
   234→  const focusRef = useRef<{ tabId: string; el: HTMLElement } | null>(null);
   235→  const trackedDocs = useRef<WeakSet<Document>>(new WeakSet());
   236→  const initedRef = useRef(false);
   237→  const lastInitialRef = useRef<string | undefined>(undefined);
   238→
   239→  const app = useMemo(() => (appSlug ? apps.find((a) => a.slug === appSlug) : undefined), [apps, appSlug]);
   240→  const active = tabs.find((t) => t.id === activeId) ?? null;
   241→  const activeRead: ScreenRead | null = active ? reads[active.id] ?? null : null;
   242→
   243→  // ── init tabs on first open ──
   244→  useEffect(() => {
   245→    if (!open || initedRef.current) return;
   246→    initedRef.current = true;
   247→    let restored: BrowserTab[] = [];
   248→    try {
   249→      const saved = JSON.parse(window.localStorage.getItem(TABS_LS) ?? "[]") as { url: string; title: string }[];
   250→      restored = saved
   251→        .filter((s) => s.url && s.url.startsWith("http"))
   252→        .slice(0, 6)
   253→        .map((s) => ({
   254→          id: uid(),
   255→          url: s.url,
   256→          title: s.title || hostOf(s.url),
   257→          loading: true,
   258→          history: [s.url],
   259→          hIdx: 0,
   260→        }));
   261→    } catch {
   262→      restored = [];
   263→    }
   264→    if (initialUrl) {
   265→      restored = restored.filter((t) => unwrapUrl(t.url) !== initialUrl);
   266→      restored.unshift({ id: uid(), url: initialUrl, title: appName || hostOf(initialUrl), loading: true, history: [initialUrl], hIdx: 0 });
   267→    }
   268→    if (restored.length === 0) {
   269→      restored = [{ id: uid(), url: "", title: "Start", loading: false, history: [], hIdx: -1 }];
   270→    }
   271→    setTabs(restored);
   272→    setActiveId(restored[0].id);
   273→    setAddr(restored[0].url === "" ? "" : restored[0].url);
   274→  }, [open, initialUrl, appName]);
   275→
   276→  // follow a NEW initialUrl while open (e.g. another kit's portal)
   277→  useEffect(() => {
   278→    if (!open || !initedRef.current) return;
   279→    if (initialUrl && initialUrl !== lastInitialRef.current) {
   280→      lastInitialRef.current = initialUrl;
   281→      setTabs((ts) => {
   282→        const existing = ts.find((t) => t.url === initialUrl);
   283→        if (existing) {
   284→          setActiveId(existing.id);
   285→          return ts;
   286→        }
   287→        const nt: BrowserTab = { id: uid(), url: initialUrl, title: appName || hostOf(initialUrl), loading: true, history: [initialUrl], hIdx: 0 };
   288→        setActiveId(nt.id);
   289→        return [...ts, nt];
   290→      });
   291→    }
   292→  }, [open, initialUrl, appName]);
   293→
   294→  // persist tab urls
   295→  useEffect(() => {
   296→    if (!open) return;
   297→    try {
   298→      window.localStorage.setItem(
   299→        TABS_LS,
   300→        JSON.stringify(tabs.filter((t) => t.url).map((t) => ({ url: t.url, title: t.title })))
   301→      );
   302→    } catch {
   303→      /* noop */
   304→    }
   305→  }, [tabs, open]);
   306→
   307→  // ── navigation ──
   308→  const navigate = useCallback(
   309→    (tabId: string, url: string, push = true) => {
   310→      setTabs((ts) =>
   311→        ts.map((t) => {
   312→          if (t.id !== tabId) return t;
   313→          const history = push ? [...t.history.slice(0, t.hIdx + 1), url] : t.history;
   314→          return { ...t, url, loading: !!url, title: url ? t.title : "Start", history, hIdx: push ? history.length - 1 : t.hIdx };
   315→        })
   316→      );
   317→      setAddr(url === "" ? "" : url);
   318→      if (url) {
   319→        const timer = setTimeout(() => {
   320→          setTabs((ts) => ts.map((t) => (t.id === tabId ? { ...t, loading: false } : t)));
   321→        }, 15000);
   322→        void timer;
   323→      }
   324→    },
   325→    []
   326→  );
   327→
   328→  const go = (delta: number) => {
   329→    if (!active) return;
   330→    const idx = active.hIdx + delta;
   331→    if (idx < 0 || idx >= active.history.length) return;
   332→    navigate(active.id, active.history[idx], false);
   333→  };
   334→
   335→  const reload = () => {
   336→    if (!active?.url) return;
   337→    try {
   338→      iframeRefs.current[active.id]?.contentWindow?.location.reload();
   339→    } catch {
   340→      setReloadKeys((rk) => ({ ...rk, [active.id]: (rk[active.id] ?? 0) + 1 }));
   341→    }
   342→  };
   343→
   344→  const addTab = (url = "") => {
   345→    const nt: BrowserTab = { id: uid(), url, title: url ? hostOf(url) : "Start", loading: !!url, history: url ? [url] : [], hIdx: url ? 0 : -1 };
   346→    setTabs((ts) => [...ts, nt]);
   347→    setActiveId(nt.id);
   348→    setAddr(url);
   349→  };
   350→
   351→  const closeTab = (id: string) => {
   352→    setTabs((ts) => {
   353→      const idx = ts.findIndex((t) => t.id === id);
   354→      const next = ts.filter((t) => t.id !== id);
   355→      if (id === activeId && next.length > 0) {
   356→        const nb = next[Math.min(idx, next.length - 1)];
   357→        setActiveId(nb.id);
   358→        setAddr(nb.url);
   359→      }
   360→      return next.length > 0 ? next : [{ id: uid(), url: "", title: "Start", loading: false, history: [], hIdx: -1 }];
   361→    });
   362→  };
   363→
   364→  // ── focus tracking in page (for "fill focused field") ──
   365→  const attachFocusTracker = useCallback((tabId: string, doc: Document) => {
   366→    if (trackedDocs.current.has(doc)) return;
   367→    trackedDocs.current.add(doc);
   368→    doc.addEventListener(
   369→      "focusin",
   370→      (e) => {
   371→        const el = e.target as HTMLElement;
   372→        if (el && /^(input|textarea|select)$/i.test(el.tagName)) focusRef.current = { tabId, el };
   373→      },
   374→      true
   375→    );
   376→  }, []);
   377→
   378→  // ── screen reader ──
   379→  const readScreen = useCallback(
   380→    (tabId: string): ScreenRead | null => {
   381→      const ifr = iframeRefs.current[tabId];
   382→      const doc = ifr?.contentDocument;
   383→      if (!doc || !doc.body) return null;
   384→      try {
   385→        const els = Array.from(doc.querySelectorAll("input, textarea, select")) as HTMLElement[];
   386→        const fields: PageField[] = [];
   387→        for (const el of els) {
   388→          const tag = el.tagName.toLowerCase() as PageField["tag"];
   389→          const type = tag === "input" ? ((el as HTMLInputElement).type || "text").toLowerCase() : tag;
   390→          if (type === "hidden" || type === "submit" || type === "button" || type === "reset" || type === "image") continue;
   391→          let id = el.getAttribute("data-sgfid");
   392→          if (!id) {
   393→            fieldCounter += 1;
   394→            id = `f${fieldCounter}`;
   395→            el.setAttribute("data-sgfid", id);
   396→          }
   397→          const rect = (el as HTMLElement).getBoundingClientRect();
   398→          const visible = rect.width > 0 && rect.height > 0;
   399→          if (!visible && type !== "file") continue;
   400→          fields.push({
   401→            id,
   402→            tag,
   403→            type,
   404→            label: findLabelText(el, doc),
   405→            name: el.getAttribute("name") ?? "",
   406→            required: el.hasAttribute("required") || el.getAttribute("aria-required") === "true",
   407→            options:
   408→              tag === "select"
   409→                ? Array.from((el as HTMLSelectElement).options)
   410→                    .slice(0, 12)
   411→                    .map((o) => o.text)
   412→                    .filter((t) => t.trim())
   413→                : [],
   414→          });
   415→        }
   416→        const bodyText = (doc.querySelector("main") ?? doc.body).innerText || "";
   417→        const captcha = !!doc.querySelector(
   418→          "iframe[src*='recaptcha'], iframe[src*='hcaptcha'], .g-recaptcha, .h-captcha, .cf-turnstile, [class*='captcha'], [id*='captcha'], [class*='turnstile'], [class*='geetest']"
   419→        );
   420→        const fileInputs = doc.querySelectorAll("input[type='file']").length;
   421→        const passwordInputs = doc.querySelectorAll("input[type='password']").length;
   422→        const payment = /card\s*number|\bcvv\b|payment|checkout|application\s*fee|billing/i.test(bodyText.slice(0, 5000));
   423→        const headings = Array.from(doc.querySelectorAll("h1, h2, h3"))
   424→          .slice(0, 8)
   425→          .map((h) => (h.textContent ?? "").trim())
   426→          .filter((t) => t && t.length < 140);
   427→        const read: ScreenRead = {
   428→          url: unwrapUrl(doc.location?.href ?? ""),
   429→          title: doc.title || "",
   430→          text: bodyText.replace(/\s+/g, " ").trim().slice(0, 6000),
   431→          headings,
   432→          fields,
   433→          captcha,
   434→          fileInputs,
   435→          passwordInputs,
   436→          payment,
   437→          readAt: Date.now(),
   438→        };
   439→        attachFocusTracker(tabId, doc);
   440→        return read;
   441→      } catch {
   442→        return null;
   443→      }
   444→    },
   445→    [attachFocusTracker]
   446→  );
   447→
   448→  const runRead = useCallback(
   449→    (tabId: string) => {
   450→      const r = readScreen(tabId);
   451→      if (r) setReads((rs) => ({ ...rs, [tabId]: r }));
   452→      return r;
   453→    },
   454→    [readScreen]
   455→  );
   456→
   457→  // ── iframe load handler ──
   458→  const handleLoad = useCallback(
   459→    (tabId: string) => {
   460→      const ifr = iframeRefs.current[tabId];
   461→      if (!ifr) return;
   462→      let realUrl = "";
   463→      let title = "";
   464→      try {
   465→        realUrl = unwrapUrl(ifr.contentWindow?.location.href ?? "");
   466→        title = ifr.contentDocument?.title || hostOf(realUrl);
   467→      } catch {
   468→        /* cross-origin — keep old */
   469→      }
   470→      setTabs((ts) =>
   471→        ts.map((t) => (t.id === tabId ? { ...t, loading: false, ...(realUrl ? { url: realUrl, title } : {}) } : t))
   472→      );
   473→      if (realUrl) pushBrowserHistory(realUrl, title);
   474→      setTimeout(() => runRead(tabId), 350);
   475→    },
   476→    [pushBrowserHistory, runRead]
   477→  );
   478→
   479→  // ── injected-script messages (SPA nav, new tabs) ──
   480→  useEffect(() => {
   481→    if (!open) return;
   482→    const onMsg = (e: MessageEvent) => {
   483→      if (e.origin !== window.location.origin) return;
   484→      const d = e.data as { source?: string; type?: string; url?: string; title?: string };
   485→      if (!d || d.source !== "sg-copilot-page") return;
   486→      const entry = Object.entries(iframeRefs.current).find(([, f]) => f?.contentWindow === e.source);
   487→      const tabId = entry?.[0];
   488→      if (!tabId) return;
